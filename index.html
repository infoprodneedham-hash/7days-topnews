async function refreshAllData() { try { const response = await fetch('/api/hub-data'); const data = await response.json(); // Update Weather document.getElementById('main-temp').textContent = `${Math.round(data.weather.main.temp)}°C`; document.getElementById('weather-desc').textContent = data.weather.weather[0].description; document.getElementById('weather-details').innerHTML = `Humidity: ${data.weather.main.humidity}% | Wind: ${data.weather.wind.speed}m/s`; // Update Forecast const forecastContainer = document.getElementById('forecast-container'); forecastContainer.innerHTML = data.forecast.map(day => { const d = new Date(day.dt * 1000).toLocaleDateString('en-AU', {weekday: 'short'}).toUpperCase(); return `
${d}
${Math.round(day.main.temp)}°
${day.weather[0].main}
`; }).join(''); // Update News const ausContainer = document.getElementById('aus-news-container'); const worldContainer = document.getElementById('world-news-container'); // Split the 8 articles into Aus and World sections const ausArticles = data.news.slice(0, 5); const worldArticles = data.news.slice(5, 8); const render = (articles) => articles.map(a => `

${a.title}

${a.description ? a.description.substring(0, 80) + '...' : 'Click to read more.'}

`).join(''); ausContainer.innerHTML = render(ausArticles); worldContainer.innerHTML = render(worldArticles); } catch (e) { console.error("Data refresh failed", e); } } // Initial load refreshAllData(); // Hourly update setInterval(refreshAllData, 3600000); // Keep your clock interval as it was setInterval(updateClocks, 1000); updateClocks();
